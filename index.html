async function callAdminFunction(action, extraData = {}) { try { const { data: { session }, error: sessionError } = await supabaseClient.auth.getSession(); if (sessionError) { console.error(sessionError); alert("Could not get your login session."); return null; } if (!session) { alert("You are not logged in."); return null; } const response = await fetch( SUPABASE_URL + "/functions/v1/admin-users", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": "Bearer " + session.access_token }, body: JSON.stringify({ action: action, ...extraData }) } ); const text = await response.text(); console.log( "Admin Function Status:", response.status ); console.log( "Admin Function Response:", text ); let data; try { data = JSON.parse(text); } catch (e) { data = { error: text }; } if (!response.ok) { alert( "Admin request failed.\n\n" + (data.error || "Server returned an error.") ); return null; } if (data.error) { alert(data.error); return null; } return data; } catch (error) { console.error( "Admin connection error:", error ); alert( "Failed to connect to the Admin Function.\n\n" + error.message ); return null; } }
